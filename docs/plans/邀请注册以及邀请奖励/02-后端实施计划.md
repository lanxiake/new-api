## 三、充值返利核心逻辑

### 3.1 服务文件

**文件**：`service/aff_rebate.go`（新建）

```go
package service

import (
    "errors"
    "fmt"
    "one-api/common"
    "one-api/model"
    "time"

    "github.com/shopspring/decimal"
    "gorm.io/gorm"
)

// CreateAffRebate 在充值成功后调用，创建一条 pending 返利记录
// 必须在 TopUp 状态已更新为 success 之后、最好放在原事务之外或同一事务的尾部
//
// 参数：
//   tx          - 可传入现有事务（推荐），nil 时使用 model.DB 新开事务
//   inviteeId   - 充值用户ID（被邀请人）
//   topupId     - 充值订单ID
//   topupTradeNo - 充值订单号
//   topupQuota  - 充值的实际 quota 数量
func CreateAffRebate(tx *gorm.DB, inviteeId int, topupId int, topupTradeNo string, topupQuota int) error {
    if !common.AffRebateEnabled {
        return nil // 功能未开启，跳过
    }
    if topupQuota <= 0 {
        return nil
    }
    if common.AffRebateRatio <= 0 {
        return nil
    }

    if tx == nil {
        tx = model.DB
    }

    // 1. 查询用户的邀请人
    var invitee model.User
    if err := tx.Select("id, inviter_id").First(&invitee, inviteeId).Error; err != nil {
        return fmt.Errorf("[CreateAffRebate] 查询用户失败: %w", err)
    }
    if invitee.InviterId == 0 {
        return nil // 该用户没有邀请人，无需返利
    }

    // 2. 计算返利金额（按 quota，向下取整）
    dQuota := decimal.NewFromInt(int64(topupQuota))
    dRatio := decimal.NewFromFloat(common.AffRebateRatio)
    rebateQuota := dQuota.Mul(dRatio).IntPart()
    if rebateQuota <= 0 {
        return nil
    }

    // 3. 计算解冻时间
    now := time.Now().Unix()
    settleAt := now + int64(common.AffRebateWaitDays)*86400

    // 4. 创建返利记录
    record := &model.AffRebateRecord{
        InviterId:    invitee.InviterId,
        InviteeId:    inviteeId,
        TopupId:      topupId,
        TopupTradeNo: topupTradeNo,
        TopupQuota:   topupQuota,
        RebateQuota:  int(rebateQuota),
        RebateRatio:  common.AffRebateRatio,
        Status:       string(model.AffRebateStatusPending),
        CreatedAt:    now,
        SettleAt:     settleAt,
    }
    if err := tx.Create(record).Error; err != nil {
        return fmt.Errorf("[CreateAffRebate] 创建返利记录失败: %w", err)
    }

    // 5. 更新邀请人的 AffPendingQuota
    if err := tx.Model(&model.User{}).Where("id = ?", invitee.InviterId).
        Update("aff_pending_quota", gorm.Expr("aff_pending_quota + ?", rebateQuota)).Error; err != nil {
        return fmt.Errorf("[CreateAffRebate] 更新待到账金额失败: %w", err)
    }

    return nil
}
```

### 3.3 防重入保护

`aff_rebate_records` 表层面无唯一索引保护，但因为：

## 四、定时结算服务

### 4.1 服务文件

**文件**：`service/aff_rebate.go`（在同一文件中追加）

```go
// SettleAffRebates 扫描到期的 pending 记录并结算
// 每次最多处理 batchSize 条
func SettleAffRebates(batchSize int) (settled int, err error) {
    if batchSize <= 0 {
        batchSize = 500
    }
    now := time.Now().Unix()

    var records []model.AffRebateRecord
    if err := model.DB.Where("status = ? AND settle_at <= ?", model.AffRebateStatusPending, now).
        Order("id ASC").Limit(batchSize).Find(&records).Error; err != nil {
        return 0, err
    }

    for _, rec := range records {
        if err := settleOneRebate(&rec); err != nil {
            common.SysLog(fmt.Sprintf("[SettleAffRebates] 结算失败 id=%d: %v", rec.Id, err))
            continue
        }
        settled++
    }
    return settled, nil
}

func settleOneRebate(rec *model.AffRebateRecord) error {
    return model.DB.Transaction(func(tx *gorm.DB) error {
        // 1. 锁定该条记录，再次校验状态（防并发）
        var current model.AffRebateRecord
        if err := tx.Set("gorm:query_option", "FOR UPDATE").
            First(&current, rec.Id).Error; err != nil {
            return err
        }
        if current.Status != string(model.AffRebateStatusPending) {
            return nil // 已被处理
        }

        // 2. 更新状态
        now := time.Now().Unix()
        if err := tx.Model(&current).Updates(map[string]interface{}{
            "status":     string(model.AffRebateStatusSettled),
            "settled_at": now,
        }).Error; err != nil {
            return err
        }

        // 3. 转移用户金额：AffPendingQuota -= rebate, AffQuota += rebate, AffHistoryQuota += rebate
        if err := tx.Model(&model.User{}).Where("id = ?", current.InviterId).
            Updates(map[string]interface{}{
                "aff_pending_quota": gorm.Expr("aff_pending_quota - ?", current.RebateQuota),
                "aff_quota":         gorm.Expr("aff_quota + ?", current.RebateQuota),
                "aff_history_quota": gorm.Expr("aff_history_quota + ?", current.RebateQuota),
            }).Error; err != nil {
            return err
        }

        // 4. 写日志
        model.RecordLog(current.InviterId, model.LogTypeSystem,
            fmt.Sprintf("邀请返利到账：来自用户#%d 的充值返利 %s",
                current.InviteeId, common.LogQuota(current.RebateQuota)))
        return nil
    })
}

// StartAffRebateScheduler 启动后台定时结算 goroutine
func StartAffRebateScheduler() {
    go func() {
        ticker := time.NewTicker(1 * time.Hour)
        defer ticker.Stop()
        for {
            // 启动时立即跑一次（处理服务停机期间到期的记录）
            n, err := SettleAffRebates(500)
            if err != nil {
                common.SysLog(fmt.Sprintf("[AffRebateScheduler] 结算异常: %v", err))
            } else if n > 0 {
                common.SysLog(fmt.Sprintf("[AffRebateScheduler] 本轮结算 %d 条", n))
            }
            <-ticker.C
        }
    }()
}
```

```go
func GetAffInvitees(c *gin.Context) {
    userId := c.GetInt("id")
    page := common.GetPageQuery(c)

    // 查询每个被邀请人的累计贡献
    type Row struct {
        InviteeId   int `json:"invitee_id"`
        TotalRebate int `json:"total_rebate"`
    }
    var rows []Row
    var total int64
    model.DB.Raw(`
        SELECT invitee_id, SUM(rebate_quota) AS total_rebate
        FROM aff_rebate_records
        WHERE inviter_id = ? AND status IN (?, ?)
        GROUP BY invitee_id
        ORDER BY total_rebate DESC
        LIMIT ? OFFSET ?
    `, userId, model.AffRebateStatusPending, model.AffRebateStatusSettled,
        page.PageSize, page.GetStartIdx()).Scan(&rows)

    // 总数（用于分页）
    model.DB.Raw(`
        SELECT COUNT(DISTINCT invitee_id) FROM aff_rebate_records
        WHERE inviter_id = ? AND status IN (?, ?)
    `, userId, model.AffRebateStatusPending, model.AffRebateStatusSettled).Scan(&total)

    // 取脱敏用户名
    inviteeIds := make([]int, len(rows))
    for i, r := range rows {
        inviteeIds[i] = r.InviteeId
    }
    userMap := make(map[int]string)
    if len(inviteeIds) > 0 {
        var users []model.User
        model.DB.Select("id, username").Where("id IN ?", inviteeIds).Find(&users)
        for _, u := range users {
            userMap[u.Id] = maskUsername(u.Username)
        }
    }

    items := make([]gin.H, len(rows))
    for i, r := range rows {
        items[i] = gin.H{
            "username":     userMap[r.InviteeId],
            "total_rebate": r.TotalRebate,
        }
    }
    page.Total = int(total)
    page.Items = items
    common.ApiSuccess(c, page)
}

func maskUsername(name string) string {
    if len(name) <= 2 { return name }
    if len(name) <= 4 { return name[:1] + "**" + name[len(name)-1:] }
    return name[:2] + "**" + name[len(name)-2:]
}
```

### 6.5 GetAffReport 实现要点（管理员）

```go
func GetAffReport(c *gin.Context) {
    var stats struct {
        TotalRebate    int `json:"total_rebate"`
        PendingRebate  int `json:"pending_rebate"`
        SettledRebate  int `json:"settled_rebate"`
        TotalInviters  int `json:"total_inviters"`
        TotalInvitees  int `json:"total_invitees"`
        TotalRecords   int `json:"total_records"`
    }
    model.DB.Raw(`
        SELECT
            COALESCE(SUM(rebate_quota), 0) AS total_rebate,
            COALESCE(SUM(CASE WHEN status='pending' THEN rebate_quota ELSE 0 END), 0) AS pending_rebate,
            COALESCE(SUM(CASE WHEN status='settled' THEN rebate_quota ELSE 0 END), 0) AS settled_rebate,
            COUNT(DISTINCT inviter_id) AS total_inviters,
            COUNT(DISTINCT invitee_id) AS total_invitees,
            COUNT(*) AS total_records
        FROM aff_rebate_records
    `).Scan(&stats)

    // TOP 10 邀请人
    var topInviters []struct {
        InviterId int    `json:"inviter_id"`
        Username  string `json:"username"`
        Total     int    `json:"total"`
    }
    model.DB.Raw(`
        SELECT r.inviter_id, u.username, SUM(r.rebate_quota) AS total
        FROM aff_rebate_records r
        LEFT JOIN users u ON u.id = r.inviter_id
        GROUP BY r.inviter_id, u.username
        ORDER BY total DESC
        LIMIT 10
    `).Scan(&topInviters)

    common.ApiSuccess(c, gin.H{
        "stats":        stats,
        "top_inviters": topInviters,
    })
}
```

> ⚠️ `CASE WHEN` 三库通用，PostgreSQL 还需注意类型转换（COALESCE 自动处理）。
