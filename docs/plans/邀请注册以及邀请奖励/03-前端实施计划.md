#### 2.3.6 划转弹窗（沿用现有 TransferModal 逻辑）

```tsx
export function TransferDialog({ children, availableQuota }: Props) {
  const [open, setOpen] = useState(false)
  const [amount, setAmount] = useState('')
  const transfer = useTransferAff()

  const handleSubmit = async () => {
    const quota = parseInt(amount, 10)
    if (isNaN(quota) || quota <= 0) {
      toast.error('请输入有效金额')
      return
    }
    await transfer.mutateAsync(quota)
    setOpen(false)
    setAmount('')
  }

  return (
    <Dialog open={open} onOpenChange={setOpen}>
      <DialogTrigger asChild>{children}</DialogTrigger>
      <DialogContent>
        <DialogHeader>
          <DialogTitle>划转到余额</DialogTitle>
          <DialogDescription>
            可用余额：{formatQuotaAsCurrency(availableQuota)}
          </DialogDescription>
        </DialogHeader>
        <Input
          type='number'
          value={amount}
          onChange={(e) => setAmount(e.target.value)}
          placeholder='请输入划转金额（quota）'
        />
        <DialogFooter>
          <Button variant='outline' onClick={() => setOpen(false)}>取消</Button>
          <Button onClick={handleSubmit} disabled={transfer.isPending}>确认</Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}
```

```tsx
import { z } from 'zod'
import { useUpdateOption } from '../hooks/use-update-option'
import { useSettingsForm } from '../hooks/use-settings-form'

const inviteSchema = z.object({
  AffRegisterRequired: z.boolean(),
  AffRebateEnabled: z.boolean(),
  AffRebateRatio: z.coerce.number().min(0).max(1),
  AffRebateWaitDays: z.coerce.number().int().min(0).max(365),
  QuotaForNewUser: z.coerce.number().int().min(0),
  QuotaForInviter: z.coerce.number().int().min(0),
  QuotaForInvitee: z.coerce.number().int().min(0),
})

export function InviteSettingsSection() {
  const { t } = useTranslation()
  const updateOption = useUpdateOption()
  const { form, onSubmit } = useSettingsForm({
    schema: inviteSchema,
    onSubmit: async (_, changedFields) => {
      for (const [k, v] of Object.entries(changedFields)) {
        await updateOption.mutateAsync({ key: k, value: v as any })
      }
    },
  })

  return (
    <SettingsCard title={t('Invitation & Rebate Settings')}>
      <Form {...form}>
        <form onSubmit={form.handleSubmit(onSubmit)} className='space-y-6'>
          {/* 邀请注册开关 */}
          <FormField name='AffRegisterRequired' render={({ field }) => (
            <FormItem className='flex items-center justify-between'>
              <div>
                <FormLabel>{t('Require Invitation Code for Registration')}</FormLabel>
                <FormDescription>
                  {t('When enabled, new users must provide a valid invitation code to register')}
                </FormDescription>
              </div>
              <Switch checked={field.value} onCheckedChange={field.onChange} />
            </FormItem>
          )}/>

          {/* 充值返利开关 */}
          <FormField name='AffRebateEnabled' render={({ field }) => (
            <FormItem className='flex items-center justify-between'>
              <div>
                <FormLabel>{t('Enable Topup Rebate')}</FormLabel>
                <FormDescription>
                  {t('When enabled, every topup of an invitee triggers rebate to the inviter')}
                </FormDescription>
              </div>
              <Switch checked={field.value} onCheckedChange={field.onChange} />
            </FormItem>
          )}/>

          {/* 返利比例 */}
          <FormField name='AffRebateRatio' render={({ field }) => (
            <FormItem>
              <FormLabel>{t('Rebate Ratio')}</FormLabel>
              <FormControl>
                <Input type='number' step='0.01' min='0' max='1' {...field} />
              </FormControl>
              <FormDescription>
                {t('Range 0-1, e.g. 0.10 means 10%')}
              </FormDescription>
            </FormItem>
          )}/>

          {/* 解冻天数 */}
          <FormField name='AffRebateWaitDays' render={({ field }) => (
            <FormItem>
              <FormLabel>{t('Settle Wait Days')}</FormLabel>
              <FormControl>
                <Input type='number' min='0' max='365' {...field} />
              </FormControl>
              <FormDescription>
                {t('Days to wait before pending rebate is settled')}
              </FormDescription>
            </FormItem>
          )}/>

          {/* 旧的注册赠送配置（保留） */}
          <FormField name='QuotaForNewUser' /* ... */ />
          <FormField name='QuotaForInviter' /* ... */ />
          <FormField name='QuotaForInvitee' /* ... */ />

          <Button type='submit'>{t('Save')}</Button>
        </form>
      </Form>
    </SettingsCard>
  )
}
```
